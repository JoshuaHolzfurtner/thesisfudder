\subsection*{Verkoerperung: Koerperzugehoerigkeit}

\begin{enumerate}
    \item Es fuehlte sich so an, als waere der virtuelle Koerper mein Koerper.
    \item Es fuehlte sich so an, als waeren die virtuellen Koerperteile meine Koerperteile.
    \item Der virtuelle Koerper fuehlte sich wie ein menschlicher Koerper an.
    \item Es fuehlte sich so an, als gehoere der virtuelle Koerper zu mir.
\end{enumerate}

\subsection*{Verkoerperung: Handlungsgefuehl}

\begin{enumerate}
    \item Die Bewegungen des virtuellen Koerpers fuehlten sich wie meine Bewegungen an.
    \item Ich hatte das Gefuehl, die Bewegungen des virtuellen Koerpers zu kontrollieren.
    \item Ich hatte das Gefuehl, die Bewegungen des virtuellen Koerpers zu verursachen.
    \item Die Bewegungen des virtuellen Koerpers waren mit meinen eigenen Bewegungen synchron.
\end{enumerate}

\subsection*{Explorativ: Koerperort}

\begin{enumerate}
    \item Ich hatte das Gefuehl, dass mein Koerper dort war, wo ich den virtuellen Ruderkoerper sah.
    \item Ich hatte das Gefuehl, dass meine Haende dort waren, wo ich die virtuellen Haende sah.
\end{enumerate}

\subsection*{Ruderspezifische Items}

\begin{enumerate}
    \item Die virtuelle Ruderbewegung passte zu dem, was ich koerperlich gefuehlt habe.
    \item Die virtuellen Haende, der Griff oder die Ruder fuehlten sich mit meiner echten Ruderbewegung verbunden an.
    \item Die virtuelle Bewegung reagierte zum richtigen Zeitpunkt auf mein Rudern.
    \item Die Ruderbewegung sah realistisch als Rudern aus.
    \item Die virtuelle Ruderszene fuehlte sich als Rudersituation glaubwuerdig an.
    \item Meine koerperliche Anstrengung fuehlte sich mit der Bewegung des virtuellen Ruderers oder Bootes verbunden an.
    \item Etwas an der Bewegung fuehlte sich unangenehm, kuenstlich oder nicht verbunden an.
\end{enumerate}
